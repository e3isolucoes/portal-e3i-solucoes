function e3iAdminCentralClone(value) { return JSON.parse(JSON.stringify(value)); }
function e3iAdminCentralSafeText(value, maxLength, fallback = '') {
  const text = String(value ?? fallback).trim();
  if (!text || text.length > maxLength || /[\u0000-\u001f\u007f]/.test(text)) return null;
  return text;
}
function e3iAdminCentralReadJson(file, fallback) {
  try { return JSON.parse(e3iAdminCentralFs.readFileSync(file, 'utf8')); }
  catch (error) { if (error && error.code === 'ENOENT') return e3iAdminCentralClone(fallback); throw error; }
}
function e3iAdminCentralWriteJsonAtomic(file, value) {
  const dir = e3iAdminCentralPath.dirname(file); e3iAdminCentralFs.mkdirSync(dir, { recursive: true, mode: 0o700 });
  const temp = `${file}.${process.pid}.${Date.now()}.tmp`; const backup = `${file}.bak`; const serialized = `${JSON.stringify(value, null, 2)}\n`; let fd;
  try {
    fd = e3iAdminCentralFs.openSync(temp, 'w', 0o600); e3iAdminCentralFs.writeFileSync(fd, serialized, 'utf8'); e3iAdminCentralFs.fsyncSync(fd); e3iAdminCentralFs.closeSync(fd); fd = undefined;
    if (e3iAdminCentralFs.existsSync(file)) { e3iAdminCentralFs.copyFileSync(file, backup); try { e3iAdminCentralFs.chmodSync(backup, 0o600); } catch (_) {} }
    e3iAdminCentralFs.renameSync(temp, file); try { e3iAdminCentralFs.chmodSync(file, 0o600); } catch (_) {}
  } finally { if (fd !== undefined) { try { e3iAdminCentralFs.closeSync(fd); } catch (_) {} } if (e3iAdminCentralFs.existsSync(temp)) { try { e3iAdminCentralFs.unlinkSync(temp); } catch (_) {} } }
}

function e3iAdminCentralPortalUsers(dataset) {
  if (!dataset || typeof dataset !== 'object') return [];
  const candidates = [dataset.users, dataset.data?.users, dataset.tables?.users?.data];
  return candidates.find(Array.isArray) || [];
}
function e3iAdminCentralMemberships(dataset) {
  if (!dataset || typeof dataset !== 'object') return [];
  const candidates = [dataset.memberships, dataset.organizationMembers, dataset.tenantMemberships, dataset.data?.memberships, dataset.tables?.organization_memberships?.data];
  return candidates.find(Array.isArray) || [];
}
function e3iAdminCentralAuthCandidates(authResult) {
  const user = authResult?.user || authResult?.session?.user || authResult?.principal || {};
  const ids = [authResult?.userId, authResult?.actorId, user?.id, user?.userId, authResult?.session?.userId].filter(Boolean).map(String);
  const emails = [authResult?.email, user?.email, authResult?.session?.email].filter(Boolean).map((value) => String(value).trim().toLowerCase());
  return { ids, emails, user };
}
function e3iAdminCentralFindUser(authResult, dataset) {
  const candidates = e3iAdminCentralAuthCandidates(authResult);
  return e3iAdminCentralPortalUsers(dataset).find((user) => {
    const id = String(user?.id || user?.userId || ''); const email = String(user?.email || '').trim().toLowerCase();
    return (id && candidates.ids.includes(id)) || (email && candidates.emails.includes(email));
  }) || null;
}
function e3iAdminCentralRole(authResult, dataset) {
  const candidates = e3iAdminCentralAuthCandidates(authResult);
  const directRoles = [authResult?.role, authResult?.systemRole, authResult?.portalRole, authResult?.session?.role, authResult?.session?.systemRole, candidates.user?.role, candidates.user?.systemRole, candidates.user?.portalRole];
  const arrays = [authResult?.roles, candidates.user?.roles, authResult?.session?.roles].filter(Array.isArray).flat();
  if ([...directRoles, ...arrays].some((role) => String(role || '').toUpperCase() === 'E3I_ADMIN')) return 'E3I_ADMIN';
  const storedUser = e3iAdminCentralFindUser(authResult, dataset);
  return [storedUser?.role, storedUser?.systemRole, storedUser?.portalRole, ...(Array.isArray(storedUser?.roles) ? storedUser.roles : [])].some((role) => String(role || '').toUpperCase() === 'E3I_ADMIN') ? 'E3I_ADMIN' : '';
}
function e3iAdminCentralActiveOrganization(authResult, dataset) {
  const user = authResult?.user || authResult?.session?.user || authResult?.principal || {};
  const direct = [authResult?.organizationId, authResult?.activeOrganizationId, authResult?.tenantId, authResult?.orgId, authResult?.organization?.id, authResult?.tenant?.id, authResult?.session?.organizationId, authResult?.session?.activeOrganizationId, authResult?.session?.tenantId, user?.organizationId, user?.activeOrganizationId, user?.tenantId].find((value) => value !== undefined && value !== null && String(value).trim());
  if (direct) return String(direct).trim();
  const storedUser = e3iAdminCentralFindUser(authResult, dataset);
  const storedDirect = storedUser?.tenantId || storedUser?.organizationId;
  if (storedDirect) return String(storedDirect).trim();
  const candidates = e3iAdminCentralAuthCandidates(authResult);
  const organizations = [...new Set(e3iAdminCentralMemberships(dataset).filter((membership) => {
    const memberId = String(membership?.userId || membership?.actorId || membership?.memberId || ''); const memberEmail = String(membership?.email || '').trim().toLowerCase(); const status = String(membership?.status || 'ACTIVE').toUpperCase();
    return ((memberId && candidates.ids.includes(memberId)) || (memberEmail && candidates.emails.includes(memberEmail))) && !['INACTIVE', 'DISABLED'].includes(status);
  }).map((membership) => String(membership?.organizationId || membership?.tenantId || membership?.orgId || '').trim()).filter(Boolean))];
  return organizations.length === 1 ? organizations[0] : '';
}
function e3iAdminCentralRequireAdmin(req, res) {
  const authResult = validateSession(req, res); if (!authResult) return null;
  let dataset = {}; try { dataset = e3iAdminCentralReadJson(E3I_ADMIN_CENTRAL_PORTAL_DATA_FILE, {}); } catch (_) {}
  if (e3iAdminCentralRole(authResult, dataset) !== 'E3I_ADMIN') { res.status(403).json({ code: 'ADMIN_REQUIRED', error: 'Acesso administrativo necessário.' }); return null; }
  const activeOrganizationId = e3iAdminCentralActiveOrganization(authResult, dataset); const requestedOrganizationId = String(req.params?.organizationId || '').trim();
  if (!activeOrganizationId || !requestedOrganizationId || activeOrganizationId !== requestedOrganizationId) { res.status(404).json({ code: 'NOT_FOUND', error: 'Recurso não encontrado.' }); return null; }
  const storedUser = e3iAdminCentralFindUser(authResult, dataset); const candidates = e3iAdminCentralAuthCandidates(authResult);
  const actorId = String(storedUser?.id || storedUser?.userId || authResult?.userId || authResult?.actorId || candidates.user?.id || 'admin').slice(0, 160);
  return { authResult, organizationId: activeOrganizationId, actorId };
}
function e3iAdminCentralRequireWrite(req, res) {
  if (String(req.get('x-e3i-admin-request') || '') !== '1') { res.status(403).json({ code: 'ADMIN_REQUEST_HEADER_REQUIRED', error: 'Requisição administrativa inválida.' }); return false; }
  return true;
}
function e3iAdminCentralRejectSecurityContext(body) {
  const forbidden = ['tenantId', 'organizationId', 'actorId', 'userId', 'id', 'password', 'passwordHash', 'roleId', 'roles', 'permissions', 'auth', 'context'];
  return body && typeof body === 'object' && forbidden.some((key) => Object.prototype.hasOwnProperty.call(body, key));
}

function e3iAdminCentralValidateSettings(candidate) {
  if (!candidate || typeof candidate !== 'object' || Array.isArray(candidate)) throw Object.assign(new Error('Configuração inválida.'), { status: 400, code: 'INVALID_SETTINGS' });
  if (Object.keys(candidate).some((key) => !['intelligence', 'governance'].includes(key))) throw Object.assign(new Error('Parâmetro não permitido.'), { status: 400, code: 'UNKNOWN_PARAMETER' });
  const intelligence = candidate.intelligence; const governance = candidate.governance;
  if (!intelligence || typeof intelligence !== 'object' || Array.isArray(intelligence) || !governance || typeof governance !== 'object' || Array.isArray(governance)) throw Object.assign(new Error('Blocos de configuração obrigatórios.'), { status: 400, code: 'INVALID_SETTINGS' });
  const intelligenceKeys = ['enabled', 'ingestionEnabled', 'agentMode', 'requireHumanApproval', 'allowSensitivePersonalData', 'defaultRetentionClass', 'mappingPurposeId']; const governanceKeys = ['auditLevel', 'savingValidationRequired'];
  if (Object.keys(intelligence).some((key) => !intelligenceKeys.includes(key)) || Object.keys(governance).some((key) => !governanceKeys.includes(key))) throw Object.assign(new Error('Parâmetro não permitido.'), { status: 400, code: 'UNKNOWN_PARAMETER' });
  for (const field of ['enabled', 'ingestionEnabled', 'requireHumanApproval', 'allowSensitivePersonalData']) if (typeof intelligence[field] !== 'boolean') throw Object.assign(new Error(`Tipo inválido para ${field}.`), { status: 400, code: 'INVALID_PARAMETER_TYPE' });
  if (!['DISABLED', 'READ_ONLY'].includes(intelligence.agentMode)) throw Object.assign(new Error('agentMode inválido.'), { status: 400, code: 'INVALID_PARAMETER_VALUE' });
  if (!['EPHEMERAL_30D', 'OPERATIONAL_1Y', 'ANALYTICS_2Y', 'AUDIT_POLICY', 'CUSTOM_POLICY'].includes(intelligence.defaultRetentionClass)) throw Object.assign(new Error('defaultRetentionClass inválido.'), { status: 400, code: 'INVALID_PARAMETER_VALUE' });
  const purpose = e3iAdminCentralSafeText(intelligence.mappingPurposeId, 120); if (!purpose || purpose.includes('@') || /\d{3}\.\d{3}\.\d{3}/.test(purpose)) throw Object.assign(new Error('mappingPurposeId inválido.'), { status: 400, code: 'INVALID_PARAMETER_VALUE' });
  if (!['STANDARD', 'ENHANCED'].includes(governance.auditLevel) || typeof governance.savingValidationRequired !== 'boolean') throw Object.assign(new Error('Governança inválida.'), { status: 400, code: 'INVALID_PARAMETER_VALUE' });
  return { intelligence: { enabled: intelligence.enabled, ingestionEnabled: intelligence.ingestionEnabled, agentMode: intelligence.agentMode, requireHumanApproval: intelligence.requireHumanApproval, allowSensitivePersonalData: intelligence.allowSensitivePersonalData, defaultRetentionClass: intelligence.defaultRetentionClass, mappingPurposeId: purpose }, governance: { auditLevel: governance.auditLevel, savingValidationRequired: governance.savingValidationRequired } };
}
function e3iAdminCentralRoot() { const root = e3iAdminCentralReadJson(E3I_ADMIN_CENTRAL_SETTINGS_FILE, { schemaVersion: 1, organizations: {} }); return root && root.schemaVersion === 1 && root.organizations && typeof root.organizations === 'object' ? root : { schemaVersion: 1, organizations: {} }; }
function e3iAdminCentralCurrent(root, organizationId) { const current = root.organizations[organizationId]; if (!current || typeof current !== 'object') return { version: 0, settings: e3iAdminCentralClone(E3I_ADMIN_CENTRAL_DEFAULTS), updatedAt: '', audit: [] }; return { version: Number.isInteger(current.version) && current.version >= 0 ? current.version : 0, settings: e3iAdminCentralValidateSettings(current.settings || e3iAdminCentralClone(E3I_ADMIN_CENTRAL_DEFAULTS)), updatedAt: String(current.updatedAt || ''), audit: Array.isArray(current.audit) ? current.audit.slice(-100) : [] }; }
function e3iAdminCentralChangedKeys(previous, next) { const keys = []; for (const [group, fields] of Object.entries(next)) for (const [key, value] of Object.entries(fields)) if (previous?.[group]?.[key] !== value) keys.push(`${group}.${key}`); return keys; }

function e3iAdminCentralLiveSessions() { return Array.isArray(sessions) ? sessions : []; }
function e3iAdminCentralLiveAudit() { return Array.isArray(auditLogs) ? auditLogs : []; }
function e3iAdminCentralUserRole(user) { return String(user?.systemRole || user?.role || 'OPERATOR').toUpperCase() === 'E3I_ADMIN' ? 'E3I_ADMIN' : 'OPERATOR'; }
function e3iAdminCentralUserStatus(user) { return String(user?.status || 'ACTIVE').toUpperCase() === 'ACTIVE' ? 'ACTIVE' : 'DISABLED'; }
function e3iAdminCentralUserBelongs(user, organizationId) { return [user?.tenantId, user?.organizationId, user?.activeOrganizationId].some((value) => String(value || '') === String(organizationId)); }

function e3iAdminCentralActiveAdminCount(organizationId) { return e3iAdminCentralLiveUsers().filter((user) => e3iAdminCentralUserBelongs(user, organizationId) && e3iAdminCentralUserStatus(user) === 'ACTIVE' && e3iAdminCentralUserRole(user) === 'E3I_ADMIN').length; }
function e3iAdminCentralRevokeSessions(userId) { const now = new Date().toISOString(); let count = 0; for (const session of e3iAdminCentralLiveSessions()) { if (String(session?.userId || '') === String(userId) && !session.revokedAt) { session.revokedAt = now; count += 1; } } return count; }
function e3iAdminCentralUserDto(user, principal) {
  const id = String(user?.id || user?.userId || ''); const activeSessions = e3iAdminCentralLiveSessions().filter((session) => String(session?.userId || '') === id && !session.revokedAt).length;
  return { id, name: String(user?.name || ''), email: String(user?.email || ''), role: e3iAdminCentralUserRole(user), status: e3iAdminCentralUserStatus(user), mustChangePassword: user?.mustChangePassword === true, activeSessions, createdAt: String(user?.createdAt || ''), updatedAt: String(user?.updatedAt || ''), isCurrentActor: id === String(principal.actorId) };
}
function e3iAdminCentralAudit(principal, action, targetUserId, detail = '') {
  e3iAdminCentralLiveAudit().unshift({ id: `log-${Date.now()}-${e3iAdminCentralCrypto.randomBytes(3).toString('hex')}`, timestamp: new Date().toISOString(), actorId: principal.actorId, userId: targetUserId, action, module: 'Administração', status: 'SUCCESS', details: String(detail || '').slice(0, 180) });
}
function e3iAdminCentralSyncMembership(user, organizationId, occurredAt = new Date().toISOString()) {
  const dataset = e3iAdminCentralReadJson(E3I_ADMIN_CENTRAL_PORTAL_DATA_FILE, {});
  const memberships = dataset?.tables?.organization_memberships?.data;
  const tenants = dataset?.tables?.tenants?.data;
  if (!Array.isArray(memberships) || !Array.isArray(tenants)) throw Object.assign(new Error('Estrutura de associação da organização indisponível.'), { status: 500, code: 'MEMBERSHIP_STORE_UNAVAILABLE' });
  const userId = String(user?.id || user?.userId || '');
  const organization = String(organizationId || '');
  if (!userId || !organization) throw Object.assign(new Error('Identidade ou organização inválida.'), { status: 500, code: 'MEMBERSHIP_CONTEXT_INVALID' });
  const role = e3iAdminCentralUserRole(user);
  const status = e3iAdminCentralUserStatus(user) === 'ACTIVE' ? 'ACTIVE' : 'DISABLED';
  let membership = memberships.find((entry) => String(entry?.userId || entry?.memberId || '') === userId && String(entry?.organizationId || entry?.tenantId || '') === organization);
  if (!membership) {
    membership = { id: `mem-${e3iAdminCentralCrypto.randomUUID()}`, userId, organizationId: organization, role, status, joinedAt: occurredAt };
    memberships.push(membership);
  } else {
    membership.userId = userId; membership.organizationId = organization; membership.role = role; membership.status = status; membership.updatedAt = occurredAt;
    if (!membership.joinedAt) membership.joinedAt = occurredAt;
  }
  if (dataset.tables?.organization_memberships) dataset.tables.organization_memberships.rowsCount = memberships.length;
  if (dataset.tables?.users && Array.isArray(dataset.tables.users.data)) dataset.tables.users.rowsCount = dataset.tables.users.data.length;
  const tenant = tenants.find((entry) => String(entry?.id || '') === organization);
  if (!tenant) throw Object.assign(new Error('Organização persistida não encontrada.'), { status: 500, code: 'ORGANIZATION_STORE_NOT_FOUND' });
  tenant.usersCount = new Set(memberships.filter((entry) => String(entry?.organizationId || entry?.tenantId || '') === organization && String(entry?.status || 'ACTIVE').toUpperCase() === 'ACTIVE').map((entry) => String(entry?.userId || entry?.memberId || '')).filter(Boolean)).size;
  dataset.lastSyncAt = occurredAt;
  e3iAdminCentralWriteJsonAtomic(E3I_ADMIN_CENTRAL_PORTAL_DATA_FILE, dataset);
}
function e3iAdminCentralValidateRole(value) { const role = String(value || '').toUpperCase(); if (!E3I_ADMIN_CENTRAL_USER_ROLES.has(role)) throw Object.assign(new Error('Papel de usuário inválido.'), { status: 400, code: 'INVALID_USER_ROLE' }); return role; }
function e3iAdminCentralValidateName(value) { const name = e3iAdminCentralSafeText(value, 160); if (!name || name.length < 2) throw Object.assign(new Error('Nome de usuário inválido.'), { status: 400, code: 'INVALID_USER_NAME' }); return name; }
function e3iAdminCentralValidateEmail(value) { const email = String(value || '').trim().toLowerCase(); if (email.length > 254 || !/^[^\s@]+@[^\s@]+\.[^\s@]+$/.test(email)) throw Object.assign(new Error('E-mail inválido.'), { status: 400, code: 'INVALID_USER_EMAIL' }); return email; }
function e3iAdminCentralProtectAdmin(principal, target, nextRole, nextStatus) {
  const targetId = String(target?.id || target?.userId || '');
  if (targetId === String(principal.actorId) && (nextRole === 'OPERATOR' || nextStatus === 'DISABLED')) throw Object.assign(new Error('A própria conta administrativa não pode remover seu acesso nesta sessão.'), { status: 409, code: 'SELF_LOCKOUT_BLOCKED' });
  if (e3iAdminCentralUserRole(target) === 'E3I_ADMIN' && e3iAdminCentralUserStatus(target) === 'ACTIVE' && (nextRole === 'OPERATOR' || nextStatus === 'DISABLED') && e3iAdminCentralActiveAdminCount(principal.organizationId) <= 1) throw Object.assign(new Error('A organização precisa manter ao menos um administrador ativo.'), { status: 409, code: 'LAST_ADMIN_REQUIRED' });
}
function e3iAdminCentralUserError(res, error, fallback) { const status = Number(error?.status) || 500; if (status >= 500) console.error('[admin-central] user administration failed'); return res.status(status).json({ code: status < 500 ? String(error?.code || 'INVALID_REQUEST') : 'INTERNAL_ERROR', error: status < 500 ? String(error?.message || fallback) : fallback }); }

  try {
    const principal = e3iAdminCentralRequireAdmin(req, res); if (!principal || !e3iAdminCentralRequireWrite(req, res)) return;
    const body = req.body; if (!body || typeof body !== 'object' || Array.isArray(body)) return res.status(400).json({ code: 'INVALID_BODY', error: 'Corpo da requisição inválido.' });
    if (e3iAdminCentralRejectSecurityContext(body)) return res.status(400).json({ code: 'SECURITY_CONTEXT_FORBIDDEN', error: 'Contexto de segurança não pode ser definido pelo cliente.' });
    if (!Number.isInteger(body.expectedVersion) || body.expectedVersion < 0) return res.status(400).json({ code: 'INVALID_VERSION', error: 'Versão esperada inválida.' });
    const nextSettings = e3iAdminCentralValidateSettings(body.settings); const root = e3iAdminCentralRoot(); const current = e3iAdminCentralCurrent(root, principal.organizationId);
    if (body.expectedVersion !== current.version) return res.status(409).json({ code: 'VERSION_CONFLICT', error: 'A configuração foi alterada por outra sessão.', currentVersion: current.version });
    const changedKeys = e3iAdminCentralChangedKeys(current.settings, nextSettings); const nextVersion = current.version + 1; const updatedAt = new Date().toISOString(); const audit = [...current.audit, { action: 'CENTRAL_SETTINGS_UPDATED', version: nextVersion, actorId: principal.actorId, changedKeys, occurredAt: updatedAt }].slice(-100);
    root.organizations[principal.organizationId] = { version: nextVersion, settings: nextSettings, updatedAt, updatedByActorId: principal.actorId, audit }; e3iAdminCentralWriteJsonAtomic(E3I_ADMIN_CENTRAL_SETTINGS_FILE, root); res.set('Cache-Control', 'no-store'); return res.json({ organizationId: principal.organizationId, version: nextVersion, settings: nextSettings, updatedAt, audit });
  } catch (error) { const status = Number(error?.status) || 500; if (status >= 500) console.error('[admin-central] settings write failed'); return res.status(status).json({ code: status < 500 ? String(error?.code || 'INVALID_SETTINGS') : 'INTERNAL_ERROR', error: status < 500 ? String(error?.message || 'Configuração inválida.') : 'Não foi possível salvar os parâmetros.' }); }

app.patch('/api/admin/organizations/:organizationId/users/:userId', (req, res) => {
  try {
    const principal = e3iAdminCentralRequireAdmin(req, res); if (!principal || !e3iAdminCentralRequireWrite(req, res)) return; const body = req.body;
